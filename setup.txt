# 1. BACKEND (FastAPI + Python)
cd Back
python -m venv venv
.\venv\Scripts\Activate.ps1   # (On macOS/Linux: source venv/bin/activate)

pip install --upgrade pip
pip install -r requirements.txt

# Create .env from example and configure API keys (GROQ, SARVAM, SUPABASE):
# cp .env.example .env

# Run Backend server (runs on http://localhost:8000)
uvicorn main:app --reload --host 0.0.0.0 --port 8000


# 2. FRONTEND WEB (React + Vite)
cd ../Front
npm install

# Create .env from example if needed:
# cp .env.example .env

npm run dev                  # (runs on http://localhost:5173)


# 3. MOBILE APP (React Native + Expo)
cd ../AgrointelMobile
npm install

# Note: In src/config.js, ensure API_BASE_URL points to your machine's Wi-Fi IP (e.g. http://192.168.x.x:8000)
# so the mobile device/Expo Go can reach your local backend.

npx expo start               # Press 'a' for Android, 'i' for iOS, or scan QR code via Expo Go
